sub check_arg
{
    my $post_data;
    read(STDIN, $post_data, $ENV{'CONTENT_LENGTH'});

    foreach (split(/&/, $post_data)) {
	my @param = split(/=/);
	$POST_PARAM{$param[0]} = $param[1];

	print "Set-Cookie: $param[0]=$param[1]; expires=Tue, 1-Jan-2030 00:00:00 GMT;\n";

sub url_decode($) {
    my $str = shift;
    $str =~ tr/+/ /;
    $str =~ s/%([0-9A-Fa-f][0-9A-Fa-f])/pack('H2', $1)/eg;
    return $str;
}
